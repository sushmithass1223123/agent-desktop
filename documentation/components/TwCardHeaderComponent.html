import { Component, EventEmitter, Input, OnInit, Output, ViewEncapsulation } from '@angular/core';
import { FuseConfig } from '@fuse/types';
import { IWidget } from 'app/interfaces';
import { AotWidgetService } from '@services/aot-widget.service';

@Component({
    selector: 'tw-card-header',
    templateUrl: './tw-card-header.component.html',
    styleUrls: ['./tw-card-header.component.scss'],
    encapsulation: ViewEncapsulation.None
})
export class TwCardHeaderComponent implements OnInit {
    @Input() data: IWidget;
    @Input() fuseConfig: FuseConfig;
    @Input() widgetState: Record<string, boolean>;

    @Output() maximize = new EventEmitter();
    @Output() float = new EventEmitter();
    @Output() collapse = new EventEmitter();
    @Output() destroy = new EventEmitter();

    constructor(
        private _aotWidgetService: AotWidgetService
    ) { }

    ngOnInit(): void { }

    maximizeWidget(): void {
        this.maximize.emit();
    }

    floatWidget(): void {
        this.float.emit();
    }

    collapseWidget(): void {
        this.collapse.emit();
    }

    destroyAOTWidget(): void {
        // get widget id
        const id = this.data.ID;
        // check if id is available, then call destroy
        if (id) {
            this._aotWidgetService.destroyWidget(id);
            this.destroy.emit();
        }
    }
}

<!-- We are using material card header -->
<mat-card-header
    class="card-header"
    [ngClass]="
        (fuseConfig.layout.anchorWidget.customBackgroundColor === true && data.Config.Anchor
            ? fuseConfig.layout.anchorWidget.headerBackground
            : fuseConfig.layout.widget.customBackgroundColor === true
            ? fuseConfig.layout.widget.headerBackground
            : '') +
        ' ' +
        (widgetState.floating ? 'floating' : '') +
        ' ' +
        (widgetState.aot ? 'aot' : '')
    "
>
    <!-- We add a material icon from config of default a widget icon -->
    <div mat-card-avatar>
        <mat-icon>{{ data.Config.Icon || 'widgets' }}</mat-icon>
    </div>

    <!-- Add titile if provided in config or add widget by default -->
    <mat-card-title>{{ data.Name || 'Widget' }}</mat-card-title>

    <span class="spacer"></span>

    <!-- All the action buttons goes here -->

    <button
        *ngIf="data.Config.Actions.includes('maximize') && !widgetState.floating && !widgetState.maximized"
        mat-icon-button
        matTooltip="Maximize"
        (click)="maximizeWidget()"
    >
        <mat-icon>fullscreen</mat-icon>
    </button>

    <button
        *ngIf="data.Config.Actions.includes('float') && !widgetState.floating && !widgetState.maximized"
        mat-icon-button
        matTooltip="Float"
        (click)="floatWidget()"
    >
        <mat-icon>picture_in_picture</mat-icon>
    </button>

    <button
        *ngIf="(widgetState.floating || widgetState.maximized) && !data.Config.Class.includes('no-restore')"
        mat-icon-button
        matTooltip="Restore"
        (click)="widgetState.floating ? floatWidget() : maximizeWidget()"
    >
        <mat-icon>close_fullscreen</mat-icon>
    </button>

    <button
        *ngIf="data.Config.Actions.includes('minimize') || widgetState.floating || widgetState.maximized"
        mat-icon-button
        [matTooltip]="widgetState.collapsed ? 'Expand' : 'Collapse'"
        (click)="collapseWidget()"
    >
        <mat-icon [ngClass]="{ 'rotate-180': !widgetState.collapsed }">expand_more</mat-icon>
    </button>

    <button *ngIf="data.Config.Actions.includes('destroy')" mat-icon-button matTooltip="Close" (click)="destroyAOTWidget()">
        <mat-icon>close</mat-icon>
    </button>
</mat-card-header>
